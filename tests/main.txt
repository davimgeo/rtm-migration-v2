  Wavelet_Create(wave);

  propagation_t* prop = Propagation_Init(

  //  specs->fwi.t0
  //);

  Wavelet_Destroy(wave);
  Model_Destroy(model);
  Seismogram_Destroy(seis);
  Propagation_Destroy(prop);
  RTM_Destroy(rtm);
  FWI_Destroy(fwi);

  return 0;
}
